/**
 * Thrown by invitation operations.
 * @param code Machine-readable discriminator for the failure:
 * - `invalid_token` — no invite matches the token (or it was already used/revoked)
 * - `expired` — the invite has passed its expiry
 * - `revoked` — the invite was revoked by an admin
 * - `already_accepted` — the invite was already claimed
 * - `already_member` — the accepting user is already a member
 * - `bad_role` — an unknown/unrecognized role was supplied
 * @param message Human-readable explanation suitable for surfacing to the user.
 */
export class InviteError extends Error {
  constructor(
    public code:
      'invalid_token' | 'expired' | 'revoked' | 'already_accepted' | 'already_member' | 'bad_role',
    message: string,
  ) {
    super(message);
  }
}
 
const DEFAULT_TTL_DAYS = 7;
/**
 * Default lifetime of an invite before it expires.
 * @remarks Mirrors the `7` in `DEFAULT_TTL_DAYS` — the invite lives 7 days.
 */
export const INVITE_TTL_MS = DEFAULT_TTL_DAYS * 24 * 60 * 60 * 1000;
 
/**
 * Generate a new cryptographically random invite token.
 * @returns A URL-safe base64 token. Only its SHA-256 digest is stored.
 */
export function newInviteToken(): string {
  return randomBytes(32).toString('base64url');
}
 
/**
 * Create an invitation for a user to join an organization.
 * @remarks The actual permission is `members.invite`; inviting beyond the current
 * seat count is permitted, with the limit enforced only at accept time. The raw
 * token is returned to the caller and is never stored.
 * @param db The database handle.
 * @param input The operation inputs.
 * @param input.orgId The organization id.
 * @param input.actorUserId The acting user's id.
 * @param input.actorRole The acting user's role.
 * @param input.email Optional pre-committed invitee email (stored on the invite).
 * @param input.role The role to grant; must be strictly below `actorRole`.
 * @param input.ttlDays Lifetime in days before the invite expires (default 7).
 * @returns The invite id, the raw sharing token, and the expiry timestamp.
 * @throws {RbacError} with code `forbidden`, `bad_role`, or `hierarchy_violation`
 * when the actor lacks permission or the requested role is invalid/not grantable.
 */
export async function createInvite(
  db: Db,
  input: {
    orgId: string;
    actorUserId: string;
    actorRole: Role;
    email?: string;
    role: string;
    ttlDays?: number;
  },
): Promise<{ id: string; token: string; expiresAtMs: number }> {
  requirePermission(input.actorRole, 'members.invite');
  const email = input.email?.trim().toLowerCase();
  if (!isRole(input.role)) {
    throw new RbacError('bad_role', `Unknown role '${input.role}'.`);
  }
  if (!mayGrant(input.actorRole, input.role)) {
    throw new RbacError(
      'hierarchy_violation',
      `A ${input.actorRole} cannot grant the '${input.role}' role.`,
    );
  }
  const ttlDays = input.ttlDays ?? DEFAULT_TTL_DAYS;
  const token = newInviteToken();
  const invite = {
    id: crypto.randomUUID(),
    orgId: input.orgId,
    tokenHash: sha256(token),
    role: input.role,
    email: email || null,
    invitedByUserId: input.actorUserId,
    expiresAtMs: Date.now() + ttlDays * 24 * 60 * 60 * 1000,
    acceptedAtMs: null,
    acceptedByUserId: null,
    revokedAtMs: null,
    createdAtMs: Date.now(),
  };
  await db.insert(invites).values(invite);
  // NOTE: the raw token is never written to the audit log.
  await audit(db, {
    orgId: input.orgId,
    actorUserId: input.actorUserId,
    action: 'invite.created',
    targetType: 'invite',
    targetId: invite.id,
    metadata: { role: input.role, email },
  });
  return { id: invite.id, token, expiresAtMs: invite.expiresAtMs };
}
 
/**
 * List the still-pending, un-expired invites for an organization.
 * @param db The database handle.
 * @param orgId The organization id.
 * @returns Invites that are neither accepted, revoked, nor expired, newest first.
 */
export async function listPendingInvites(db: Db, orgId: string) {
  return db
    .select({
      id: invites.id,
      role: invites.role,
      email: invites.email,
      expiresAtMs: invites.expiresAtMs,
      createdAtMs: invites.createdAtMs,
    })
    .from(invites)
    .where(
      and(
        eq(invites.orgId, orgId),
        isNull(invites.acceptedAtMs),
        isNull(invites.revokedAtMs),
        gt(invites.expiresAtMs, Date.now()),
      ),
    )
    .orderBy(desc(invites.createdAtMs));
}
 
/**
 * Revoke a pending invite so it can no longer be accepted.
 * @remarks Requires `invites.revoke`. The update is conditional so already-used
 * or already-revoked invites cannot be revoked again.
 * @param db The database handle.
 * @param input The operation inputs.
 * @param input.orgId The organization id (scoping safeguard).
 * @param input.actorRole The acting user's role.
 * @param input.actorUserId The acting user's id.
 * @param input.inviteId The invite id to revoke.
 * @throws {RbacError} with code `forbidden` when the permission is missing.
 * @throws {InviteError} with code `invalid_token` when the invite is missing,
 * already used, or already revoked.
 */
export async function revokeInvite(
  db: Db,
  input: { orgId: string; actorRole: Role; actorUserId: string; inviteId: string },
): Promise<void> {
  requirePermission(input.actorRole, 'invites.revoke');
  const updated = await db
    .update(invites)
    .set({ revokedAtMs: Date.now() })
    .where(
      and(
        eq(invites.id, input.inviteId),
        eq(invites.orgId, input.orgId),
        isNull(invites.revokedAtMs),
        isNull(invites.acceptedAtMs),
      ),
    )
    .returning({ id: invites.id });
  if (updated.length === 0)
    throw new InviteError('invalid_token', 'Invite not found, already used, or already revoked.');
  await audit(db, {
    orgId: input.orgId,
    actorUserId: input.actorUserId,
    action: 'invite.revoked',
    targetType: 'invite',
    targetId: input.inviteId,
  });
}
 
/**
 * Read-only preview for the landing page of an invite link — never mutates.
 * @param db The database handle.
 * @param token The raw invite token.
 * @returns A discriminated union describing the invite's state: `invalid`,
 * `expired`, `revoked`, `accepted`, or `valid` (with org name, role, and expiry).
 */
export async function peekInvite(
  db: Db,
  token: string,
): Promise<
  | { state: 'invalid' }
  | { state: 'expired' }
  | { state: 'revoked' }
  | { state: 'accepted' }
  | { state: 'valid'; orgName: string; role: Role; expiresAtMs: number }
> {
  const invite = (
    await db
      .select()
      .from(invites)
      .where(eq(invites.tokenHash, sha256(token)))
      .limit(1)
  )[0];
  if (!invite) return { state: 'invalid' };
  if (invite.revokedAtMs) return { state: 'revoked' };
  if (invite.acceptedAtMs) return { state: 'accepted' };
  if (invite.expiresAtMs <= Date.now()) return { state: 'expired' };
  const org = (
    await db
      .select({ name: organizations.name })
      .from(organizations)
      .where(eq(organizations.id, invite.orgId))
      .limit(1)
  )[0];
  return {
    state: 'valid',
    orgName: org?.name ?? '',
    role: invite.role as Role,
    expiresAtMs: invite.expiresAtMs,
  };
}
