/**
 * Thrown by membership-management operations.
 * @param code Machine-readable discriminator for the failure:
 * - `no_membership` — the target user is not a member of the organization
 * - `self_remove` — a member tried to remove itself (use "Leave organization")
 * - `hierarchy_violation` — the actor tried to act on a peer, a superior, or itself
 * - `last_owner` — the operation would leave the organization with no owner
 * - `bad_role` — an unknown/unrecognized role was supplied
 * @param message Human-readable explanation suitable for surfacing to the user.
 */
export class MemberError extends Error {
  constructor(
    public code:
      'no_membership' | 'self_remove' | 'hierarchy_violation' | 'last_owner' | 'bad_role',
    message: string,
  ) {
    super(message);
  }
}
 
/** A member of an organization, joined with the user's public profile. */
export type MemberRow = {
  membershipId: string;
  userId: string;
  email: string;
  name: string;
  role: Role;
  createdAtMs: number;
};
 
/**
 * List all members of an organization with their roles.
 * @param db The database handle.
 * @param orgId The organization id.
 * @returns An array of `MemberRow`.
 */
export async function listMembers(db: Db, orgId: string): Promise<MemberRow[]> {
  const rows = await db
    .select({
      membershipId: memberships.id,
      userId: users.id,
      email: users.email,
      name: users.name,
      role: memberships.role,
      createdAtMs: memberships.createdAtMs,
    })
    .from(memberships)
    .innerJoin(users, eq(users.id, memberships.userId))
    .where(eq(memberships.orgId, orgId));
  return rows.map((r) => ({ ...r, role: r.role as Role }));
}
 
async function getMembershipRow(db: Db, orgId: string, userId: string) {
  return (
    await db
      .select()
      .from(memberships)
      .where(and(eq(memberships.orgId, orgId), eq(memberships.userId, userId)))
      .limit(1)
  )[0];
}
 
function countOwners(members: { role: string; userId: string }[]): number {
  return members.filter((m) => m.role === 'owner').length;
}

/**
 * Remove the acting user from an organization on their own initiative.
 * @remarks Enforces the single-owner invariant: an owner cannot leave while they
 * are the last owner.
 * @param db The database handle.
 * @param input The operation inputs.
 * @param input.orgId The organization id.
 * @param input.userId The member leaving.
 * @throws {MemberError} with code `no_membership` when the user is not a member,
 * or `last_owner` when a sole owner tries to leave.
 */
export async function leaveOrg(db: Db, input: { orgId: string; userId: string }): Promise<void> {
  const membership = await getMembershipRow(db, input.orgId, input.userId);
  if (!membership)
    throw new MemberError('no_membership', 'You are not a member of this organization.');
  if (membership.role === 'owner') {
    const all = await db
      .select({ userId: memberships.userId, role: memberships.role })
      .from(memberships)
      .where(eq(memberships.orgId, input.orgId));
    if (countOwners(all) <= 1) {
      throw new MemberError('last_owner', 'Transfer ownership before leaving the organization.');
    }
  }
  await db.delete(memberships).where(eq(memberships.id, membership.id));
  await audit(db, {
    orgId: input.orgId,
    actorUserId: input.userId,
    action: 'member.left',
    targetType: 'user',
    targetId: input.userId,
    metadata: { role: membership.role },
  });
}
